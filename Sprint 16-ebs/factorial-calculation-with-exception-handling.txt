

FACTORIAL WITHOUT EXCEPTION HANDLING
============================================= 

SQL> CREATE OR REPLACE PROCEDURE ALHARITH_PR_FACT(N NUMBER) IS
	S NUMBER :=N;
	RES NUMBER :=N;

	BEGIN

	LOOP
	S :=S-1;
	RES :=RES*S;
	EXIT WHEN S=1;

	END LOOP;

	DBMS_OUTPUT.PUT_LINE('FACTORIAL='||RES);
	END;
/

SQL> SET SERVEROUTPUT ON; 

SQL> EXEC ALHARITH_PR_FACT(5);




FACTORIAL WITH EXCEPTION HANDLING
============================================= 

SQL> CREATE OR REPLACE PROCEDURE ALHARITH_PR_FACT_UPDATED(ERRBUF OUT VARCHAR2,RETCODE OUT NUMBER, N NUMBER) IS

	S NUMBER := N;
	RES NUMBER := N;

	FACT_1 EXCEPTION;
	FACT_0 EXCEPTION;

	BEGIN
	IF S = 0 OR S = 1 THEN
	RAISE FACT_1;

	ELSIF S < 0 THEN
	RAISE FACT_0;

	END IF;

	LOOP
	S := S - 1;
	RES := RES * S;
	EXIT WHEN S = 2;
	END LOOP;

	FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' FACTORIAL = '||RES);


	EXCEPTION

	WHEN FACT_1 THEN
	FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' FACTORIAL = 1 ');

	WHEN FACT_0 THEN
	FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' INPUT N CANNOT BE LESS THAN 0 ');

	WHEN NO_DATA_FOUND THEN
	FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' INPUT A NUMBER TO COMPUTE FACTORIAL');

	WHEN OTHERS THEN
	FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' CHECK YOUR PROGRAM !!! ');

	END;
	/
